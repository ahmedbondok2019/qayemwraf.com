Route::post('/get-shipping-cost', function (Request $request) {
    $cost = \App\Models\ShippingCategoryArea::where('area_id', $request->area_id)
        ->where('shipping_category_id', $request->shipping_category_id)
        ->first()?->value ?? 50;

    return response()->json(['shipping_cost' => $cost]);
})->name('get.shipping.cost');
